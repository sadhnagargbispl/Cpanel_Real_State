<script>
    /* ── Budget slider labels ──────────────────────────────── */
    var budgetLabels = ['₹0 Lac','₹50 Lac','₹1 Cr','₹3 Cr','₹5 Cr+'];
    function updateBudgetLabel(v) {
        document.getElementById('lblBudgetVal').textContent = budgetLabels[v];
        document.getElementById('<%= hfBudgetRange.ClientID %>').value = v;
    }
    // init on load
    (function(){
        var v = parseInt(document.getElementById('<%= hfBudgetRange.ClientID %>').value || '4');
        var el = document.getElementById('rngBudget');
        if(el){ el.value = v; updateBudgetLabel(v); }
    })();

    /* ── Sync checkbox selections into hidden fields ───────── */
    function syncCheckboxes() {
        var types = [];
        document.querySelectorAll('.type-check input[type=checkbox]:checked').forEach(function(cb){
            types.push(cb.closest('label').querySelector('asp\\:CheckBox, input').getAttribute('data-typeid')
                || cb.parentElement.getAttribute('data-typeid'));
        });

        var statuses = [];
        document.querySelectorAll('.status-check input[type=checkbox]:checked').forEach(function(cb){
            statuses.push(cb.closest('label').querySelector('input[type=checkbox]').getAttribute('data-statusid')
                || cb.parentElement.getAttribute('data-statusid'));
        });

        document.getElementById('<%= hfSelectedTypes.ClientID %>').value    = types.join(',');
        document.getElementById('<%= hfSelectedStatuses.ClientID %>').value = statuses.join(',');
    }

    /* attach change listeners after DOM ready */
    document.addEventListener('DOMContentLoaded', function(){
        document.querySelectorAll('.type-check input, .status-check input').forEach(function(cb){
            cb.addEventListener('change', function(){
                syncCheckboxes();
                document.getElementById('<%= hfPage.ClientID %>').value = '1';
                __doPostBack('<%= btnSearch.UniqueID %>', '');
            });
        });

        /* restore checked state from hidden fields */
        var selTypes    = (document.getElementById('<%= hfSelectedTypes.ClientID %>').value || '').split(',');
        var selStatuses = (document.getElementById('<%= hfSelectedStatuses.ClientID %>').value || '').split(',');

        document.querySelectorAll('.type-check').forEach(function(wrap){
            var id = wrap.getAttribute('data-typeid');
            if(id && selTypes.indexOf(id) !== -1)
                wrap.querySelector('input[type=checkbox]').checked = true;
        });
        document.querySelectorAll('.status-check').forEach(function(wrap){
            var id = wrap.getAttribute('data-statusid');
            if(id && selStatuses.indexOf(id) !== -1)
                wrap.querySelector('input[type=checkbox]').checked = true;
        });
    });

    /* ── Pagination ─────────────────────────────────────────── */
    function goPage(n) {
        syncCheckboxes();
        document.getElementById('<%= hfPage.ClientID %>').value = n;
        __doPostBack('<%= btnSearch.UniqueID %>', '');
    }

    /* ── Clear all ──────────────────────────────────────────── */
    function clearFilters() {
        document.getElementById('<%= txtLocation.ClientID %>').value = '';
        document.getElementById('ddlProjectType').selectedIndex = 0;
        document.getElementById('ddlPlotSize').selectedIndex    = 0;
        document.getElementById('<%= ddlBudget.ClientID %>').selectedIndex   = 0;
        document.getElementById('<%= ddlStatus.ClientID %>').selectedIndex   = 0;
        document.getElementById('<%= hfSelectedTypes.ClientID %>').value    = '';
        document.getElementById('<%= hfSelectedStatuses.ClientID %>').value = '';
        document.getElementById('<%= hfBudgetRange.ClientID %>').value      = '4';
        document.getElementById('rngBudget').value = 4;
        updateBudgetLabel(4);
        document.getElementById('<%= hfPage.ClientID %>').value = '1';
        __doPostBack('<%= btnSearch.UniqueID %>', '');
    }

    /* ── Fade-in observer ───────────────────────────────────── */
    var obs = new IntersectionObserver(function(entries){
        entries.forEach(function(e,i){
            if(e.isIntersecting)
                setTimeout(function(){ e.target.classList.add('visible'); }, i*80);
        });
    },{ threshold:0.08 });
    document.querySelectorAll('.fade-in').forEach(function(el){ obs.observe(el); });

    /* ── Wishlist heart ─────────────────────────────────────── */
    document.querySelectorAll('.prop-save').forEach(function(btn){
        btn.addEventListener('click', function(e){
            e.stopPropagation();
            btn.textContent = btn.textContent.trim()==='♡' ? '❤️' : '♡';
        });
    });
</script>

/* ── 1. Add these two new model classes at the bottom of the file ─── */

public class TypeFilterItem
{
    public int    ProjectTypeID { get; set; }
    public string TypeName      { get; set; }
    public int    Count         { get; set; }
}

public class StatusFilterItem
{
    public string StatusID    { get; set; }
    public string StatusLabel { get; set; }
    public int    Count       { get; set; }
}


/* ── 2. Add this method to load sidebar counts ──────────────────── */

private void BindSidebarFilters()
{
    using (var con = DBHelper.GetConnection())
    {
        con.Open();

        // ── Type counts ──
        string typeSql = @"
            SELECT pt.ProjectTypeID, pt.TypeName,
                   COUNT(DISTINCT p.ProjectID) AS Count
            FROM   dbo.MstProjectTypes pt
            LEFT JOIN dbo.Projects p
                ON p.ProjectTypeID = pt.ProjectTypeID
               AND p.IsDeleted = 0 AND p.IsShowOnWebsite = 1
            WHERE  pt.IsActive = 1
            GROUP BY pt.ProjectTypeID, pt.TypeName
            ORDER BY pt.TypeName";

        var typeList = new System.Collections.Generic.List<TypeFilterItem>();
        using (var cmd = new SqlCommand(typeSql, con))
        using (var dr = cmd.ExecuteReader())
            while (dr.Read())
                typeList.Add(new TypeFilterItem
                {
                    ProjectTypeID = Convert.ToInt32(dr["ProjectTypeID"]),
                    TypeName      = dr["TypeName"].ToString(),
                    Count         = Convert.ToInt32(dr["Count"])
                });

        rptTypeFilter.DataSource = typeList;
        rptTypeFilter.DataBind();

        // ── Status counts ──
        // PublishMode stores the label ("Active","Upcoming","Pre-Launch","Sold Out")
        string statusSql = @"
            SELECT p.PublishMode AS StatusLabel,
                   COUNT(DISTINCT p.ProjectID) AS Count
            FROM   dbo.Projects p
            WHERE  p.IsDeleted = 0 AND p.IsShowOnWebsite = 1
            GROUP BY p.PublishMode
            ORDER BY p.PublishMode";

        var statusList = new System.Collections.Generic.List<StatusFilterItem>();
        using (var cmd2 = new SqlCommand(statusSql, con))
        using (var dr2 = cmd2.ExecuteReader())
            while (dr2.Read())
                statusList.Add(new StatusFilterItem
                {
                    StatusID    = dr2["StatusLabel"].ToString(),   // use label as ID
                    StatusLabel = dr2["StatusLabel"].ToString(),
                    Count       = Convert.ToInt32(dr2["Count"])
                });

        rptStatusFilter.DataSource = statusList;
        rptStatusFilter.DataBind();
    }
}
